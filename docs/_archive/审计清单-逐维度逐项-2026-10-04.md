### F20 · 配置数值校验：让"配置写错"报 2 而不是 1

```python
# lvs/config.py · Config 类（放在 require() 旁边）
    def as_int(self, dotted: str, default: int = 0) -> int:
        """取整数配置项。配置写错不是"未预期的内部错误"。

        实测：`width = "1344px"` 会让 `int(config.get(...))` 抛 ValueError，
        经 `cli.main` 兜底变成「内部错误：ValueError: invalid literal for int()」+ 退出码 1。
        退出码 1 的语义是"跑完了但有失败件" —— 自动化脚本会据此**重试**，
        而正确动作是"别重试、去改配置"（= 2 / EXIT_USAGE）。ConfigError 继承 UsageError，
        所以这里抛 ConfigError 就自动拿到了正确退出码。
        """
        raw = self.get(dotted, default)
        if raw is None or raw == "":
            raw = default
        try:
            return int(raw)
        except (TypeError, ValueError):
            where = f"{self.path}" if self.path else f"{PROJECT_ROOT / CONFIG_FILENAME}"
            raise ConfigError(f"配置项 `{dotted}` 必须是整数，实际是 {raw!r}。位置：{where}") from None

    def as_float(self, dotted: str, default: float = 0.0) -> float:
        raw = self.get(dotted, default)
        if raw is None or raw == "":
            raw = default
        try:
            return float(raw)
        except (TypeError, ValueError):
            where = f"{self.path}" if self.path else f"{PROJECT_ROOT / CONFIG_FILENAME}"
            raise ConfigError(f"配置项 `{dotted}` 必须是数字，实际是 {raw!r}。位置：{where}") from None
```

def harden(path: Path) -> None:
    """把含密钥的配置文件收紧到仅属主可读写（尽力而为，失败不阻断）。"""
    try:
        if os.name == "nt":
            subprocess.run(["icacls", str(path), "/inheritance:r",
                            "/grant:r", f"{os.environ.get('USERNAME', '')}:(R,W)"],
                           check=False, capture_output=True)
        else:
            path.chmod(0o600)
    except OSError:
        pass   # 只读盘 / 无 icacls：不因为"收不紧权限"让流程失败
